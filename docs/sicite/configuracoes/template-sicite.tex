% O arquivo .bst altera esta chave para falso quando encontra ao menos uma
% referencia real. Entradas cujas chaves comecam por "modelo-" sao apenas os
% exemplos distribuidos com o template.
\newif\ifsicitereferenciasmodelo
\sicitereferenciasmodelotrue
\renewenvironment{thebibliography}[1]
  {%
    \sicitesection[\sicitespaceAfterReferenciasTitle]{REFER\^ENCIAS}%
    \sicitefontsize
    \ifsicitereferenciasmodelo
      \siciteparagraph{%
        Deve-se utilizar no texto o formato: trabalho\textsuperscript{[1]}.
        Apesar deste par\'agrafo de instru\c{c}\~oes, nesta se\c{c}\~ao deve-se
        apresentar apenas a lista de refer\^encias como no exemplo a seguir. As
        refer\^encias devem ser alinhadas \`a esquerda.%
      }%
    \fi
    \setcounter{enumiv}{0}%
    \let\p@enumiv\@empty
    \renewcommand\theenumiv{\@arabic\c@enumiv}%
    \def\@bibitem##1{%
      \par\refstepcounter{enumiv}%
      \if@filesw
        \immediate\write\@auxout{\string\bibcite{##1}{\theenumiv}}%
      \fi
      \noindent[\theenumiv]~\ignorespaces
    }%
    \def\@lbibitem[##1]##2{%
      \par\refstepcounter{enumiv}%
      \if@filesw
        \immediate\write\@auxout{\string\bibcite{##2}{##1}}%
      \fi
      \noindent[##1]~\ignorespaces
    }%
    \sloppy
    \clubpenalty4000
    \@clubpenalty \clubpenalty
    \widowpenalty4000
    \sfcode`\.=\@m
  }
  {%
    \def\@noitemerr{\@latex@warning{Ambiente thebibliography vazio}}%
    \par
  }
\makeatother
